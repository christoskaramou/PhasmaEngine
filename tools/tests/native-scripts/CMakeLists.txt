cmake_minimum_required(VERSION 3.22)
project(NativeScriptCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(API "${ENGINE_ROOT}/Phasma/ProjectNative/Api")
set(RUNTIME "${ENGINE_ROOT}/Phasma/Runtime/Code/Script")
foreach(variant RANGE 1 7)
    add_library(probe${variant} SHARED module.cpp)
    target_include_directories(probe${variant} PRIVATE "${API}")
    target_compile_definitions(probe${variant} PRIVATE PROBE_VARIANT=${variant})
    if(MSVC)
        target_link_options(probe${variant} PRIVATE /DEBUG) # RSDS record + probeN.pdb for the PDB patch check
    elseif(CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC") # clang++ (GNU driver) targeting MSVC: -g links with /DEBUG
        target_compile_options(probe${variant} PRIVATE -g)
        target_link_options(probe${variant} PRIVATE -g)
    endif()
endforeach()
add_library(linked_probe STATIC module.cpp)
target_include_directories(linked_probe PRIVATE "${API}")
target_compile_definitions(linked_probe PRIVATE PROBE_VARIANT=1)
add_executable(native_script_check check.cpp "${RUNTIME}/ProjectNativeHooks.cpp" "${RUNTIME}/NativeScriptGuard.cpp")
find_package(Threads REQUIRED)
target_link_libraries(native_script_check PRIVATE linked_probe Threads::Threads)
target_include_directories(native_script_check PRIVATE "${API}" "${RUNTIME}" "${ENGINE_ROOT}/Phasma/Core/third_party/glm")
if(WIN32)
    target_include_directories(native_script_check PRIVATE "${ENGINE_ROOT}/Phasma/Core/third_party/SDL2")
    target_link_libraries(native_script_check PRIVATE "${ENGINE_ROOT}/Phasma/Core/Libs/SDL2/SDL2.lib")
    add_custom_command(TARGET native_script_check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ENGINE_ROOT}/Phasma/Core/DLLs/SDL2.dll" "$<TARGET_FILE_DIR:native_script_check>")
else()
    find_package(SDL2 REQUIRED)
    target_link_libraries(native_script_check PRIVATE SDL2::SDL2)
endif()
enable_testing()
add_library(sample_game SHARED "${ENGINE_ROOT}/Phasma/ProjectNative/Module.cpp"
    "${ENGINE_ROOT}/Phasma/ProjectNative/Sample/Orbit.cpp"
    "${ENGINE_ROOT}/Phasma/ProjectNative/Sample/PlayerController.cpp")
target_include_directories(sample_game PRIVATE "${API}")
include("${ENGINE_ROOT}/Phasma/ProjectNative/ProjectNative.cmake")
get_target_property(sample_sources sample_game SOURCES)
pe_project_native_name_sources(${sample_sources})
add_test(NAME native_script_reload COMMAND native_script_check
    $<TARGET_FILE:probe1> $<TARGET_FILE:probe2> $<TARGET_FILE:probe3>
    $<TARGET_FILE:probe4> $<TARGET_FILE:probe5> $<TARGET_FILE:probe6> $<TARGET_FILE:sample_game> $<TARGET_FILE:probe7>)
